<script>
let items = [];
let vouchers = JSON.parse(localStorage.getItem("hybridVouchers") || "[]");

const $ = id => document.getElementById(id);
const money = n => "৳" + Number(n || 0).toLocaleString("en-BD", {
minimumFractionDigits:2, maximumFractionDigits:2
});

function newVoucher(){
$("voucher").value = "HC-" + Date.now().toString().slice(-8);
$("date").value = new Date().toISOString().slice(0,10);
["client","phone","project","address"].forEach(id=>$ (id).value="");
$("discount").value=0;
$("paid").value=0;
items=[];
render();
}

function addItem(){
let service=$("service").value;
let qty=Number($("qty").value);
let price=Number($("price").value);

if(!service || qty<=0 || price<0 || $("price").value===""){
alert("সার্ভিস, পরিমাণ ও মূল্য দিন");
return;
}

items.push({service,qty,price});
render();
}

function removeItem(i){
items.splice(i,1);
render();
}

function render(){
$("items").innerHTML=items.map((x,i)=>`
<tr>
<td>${escapeHTML(x.service)}</td>
<td>${x.qty}</td>
<td>${money(x.price)}</td>
<td>${money(x.qty*x.price)}</td>
<td class="no-print">
<button onclick="removeItem(${i})">X</button>
</td>
</tr>
`).join("");
calculate();
}

function calculate(){
let subtotal=items.reduce((s,x)=>s+x.qty*x.price,0);
let discount=Math.max(0,Number($("discount").value)||0);
let net=Math.max(0,subtotal-discount);
let paid=Math.max(0,Number($("paid").value)||0);
let due=Math.max(0,net-paid);

$("subtotal").textContent=money(subtotal);
$("discountText").textContent=money(discount);
$("net").textContent=money(net);
$("paidText").textContent=money(paid);
$("due").textContent=money(due);
}

function getVoucher(){
return {
voucher:$("voucher").value,
date:$("date").value,
client:$("client").value,
phone:$("phone").value,
project:$("project").value,
address:$("address").value,
items,
discount:Number($("discount").value)||0,
paid:Number($("paid").value)||0
};
}

function saveVoucher(){
if(!$("client").value.trim()){
alert("ক্লায়েন্টের নাম লিখুন");
return;
}
if(!items.length){
alert("কমপক্ষে একটি সার্ভিস যোগ করুন");
return;
}

let v=getVoucher();
let i=vouchers.findIndex(x=>x.voucher===v.voucher);

if(i>=0) vouchers[i]=v;
else vouchers.push(v);

localStorage.setItem("hybridVouchers",JSON.stringify(vouchers));
updateList();
alert("Voucher সংরক্ষণ হয়েছে");
}

function updateList(){
$("savedList").innerHTML='<option value="">Voucher নির্বাচন করুন</option>'+
vouchers.map(v=>`<option value="${escapeHTML(v.voucher)}">${escapeHTML(v.voucher)} - ${escapeHTML(v.client)}</option>`).join("");
}

function loadVoucher(){
let v=vouchers.find(x=>x.voucher===$("savedList").value);
if(!v)return;

$("voucher").value=v.voucher;
$("date").value=v.date;
$("client").value=v.client;
$("phone").value=v.phone;
$("project").value=v.project;
$("address").value=v.address;
$("discount").value=v.discount;
$("paid").value=v.paid;
items=v.items;
render();
}

function deleteVoucher(){
let id=$("savedList").value;
if(!id)return;
if(!confirm("এই Voucher মুছে ফেলবেন?"))return;

vouchers=vouchers.filter(v=>v.voucher!==id);
localStorage.setItem("hybridVouchers",JSON.stringify(vouchers));
updateList();
}

function escapeHTML(s){
return String(s).replace(/[&<>"']/g,c=>({
"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
}[c]));
}

newVoucher();
updateList();
</script>
